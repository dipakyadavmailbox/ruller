---
/**
 * Nav.astro — site header.
 *
 * Every link is run through localizePath() so that a visitor browsing in
 * Spanish stays in Spanish as they move between tools, and labels come
 * from the UI dictionary rather than being hardcoded in English.
 */
import LanguageSwitcher from './LanguageSwitcher.astro'
import { CATEGORIES } from '../../data/tools.js'
import { DEFAULT_LOCALE } from '../../i18n/config.js'
import { localizePath, normalizePath } from '../../i18n/routing.js'
import { categoryLabel, toolLabel } from '../../i18n/toolKeys.js'
import { t } from '../../i18n/ui.js'

interface Props {
  lang?: string
  /** Locale-free canonical path of the current page. */
  path?: string
}

const { lang = DEFAULT_LOCALE, path } = Astro.props

// Prefer the declared path; fall back to the URL for pages that predate the
// prop. Either way this is compared locale-free against the tool slugs.
const currentPath = normalizePath(path ?? Astro.url.pathname)

const href = (target) => localizePath(target, lang)
const isActive = (target) => currentPath === target
---

<header class="site-header" id="site-header">
  <div class="header-left">
    <a href={href('/')} class="logo-link">
      <div class="logo-badge">
        <span class="logo-spark">✨</span>
      </div>
      <div class="logo-text-group">
        <span class="logo-title">Rocking Tools</span>
        <span class="logo-sub"></span>
      </div>
    </a>
  </div>

  <!-- Navigation Menu (Desktop inline, Mobile Drawer) -->
  <nav class="site-nav" id="site-nav">
    <div class="nav-content">
      <a href={href('/')} class={isActive('/') ? 'nav-item active' : 'nav-item'}>{t(lang, 'nav.allTools')}</a>

      {CATEGORIES.map((cat) => {
        const hasActive = cat.tools.some((tool) => isActive(tool.slug))
        return (
          <div class="nav-dropdown">
            <button class={hasActive ? 'dropdown-trigger active' : 'dropdown-trigger'} type="button" aria-expanded="false">
              <span>{categoryLabel(cat.heading, lang)}</span>
              <span class="arrow">▾</span>
            </button>
            <div class="dropdown-menu glass-panel">
              {cat.tools.map((tool) => (
                <a href={href(tool.slug)} class={isActive(tool.slug) ? 'dropdown-item active' : 'dropdown-item'}>
                  <span class="tool-icon">{tool.icon}</span>
                  <div class="tool-info">
                    <span class="tool-name">{toolLabel(tool.slug, tool.navLabel, lang)}</span>
                  </div>
                </a>
              ))}
            </div>
          </div>
        )
      })}

      <a href="/workspace" class={isActive('/workspace') ? 'workspace-link active' : 'workspace-link'} aria-label={t(lang, 'nav.workspace')}>🚀 {t(lang, 'nav.workspace')}</a>

      <a href={href('/about')} class={isActive('/about') ? 'nav-item active' : 'nav-item'}>{t(lang, 'nav.about')}</a>
      <a href={href('/faq')} class={isActive('/faq') ? 'nav-item active' : 'nav-item'}>{t(lang, 'nav.faq')}</a>
      <a href="/contact" class={isActive('/contact') ? 'nav-item active' : 'nav-item'}>{t(lang, 'nav.contact')}</a>
      <a href="/donate" class="support-link" aria-label={t(lang, 'nav.support')}>💛 {t(lang, 'nav.support')}</a>

      <LanguageSwitcher lang={lang} path={currentPath} />

      <!-- Theme Toggle (Placed at the very last position) -->
      <button id="theme-toggle" class="theme-toggle" aria-label={t(lang, 'nav.theme')} type="button">
        <span id="theme-toggle-label" class="theme-toggle-label" data-label-dark={t(lang, 'nav.themeDark')} data-label-light={t(lang, 'nav.themeLight')}>{t(lang, 'nav.theme')}</span>
      </button>
    </div>
  </nav>

  <!-- Mobile Hamburger Toggle Button -->
  <button id="mobile-toggle" class="mobile-toggle" aria-label={t(lang, 'nav.toggleMenu')} aria-expanded="false" aria-controls="site-nav" type="button">
    <span class="hamburger-bar"></span>
    <span class="hamburger-bar"></span>
    <span class="hamburger-bar"></span>
  </button>

  <!-- Mobile Backdrop Overlay -->
  <div class="nav-backdrop" id="nav-backdrop" aria-hidden="true"></div>
</header>
